  - item: "Accomplished Request Slip for duly registered Civil Registry Document (CRD) such as Birth, Marriage, Death, Fetal Death, Islam Conversion and other CRDs – one (1) copy for each request"
    whereToSecure: "Local Civil Registry Office/Information Desk/ Visitor's Log Book Registration Window"
  - item: "Any Government Issued Valid ID of the document owner, duly authorized person and authorizing person - (present original copy)"
    whereToSecure: "Document Owner/Authorized Person/Authorizing Person"
  - item: "Duly Notarized Special Power of Attorney (SPA) or Authorization letter, if applicable – one (1) original copy"
    whereToSecure: "Notary Public/ Document Owner/ Authorized Person/Authorizing Person (in the order mentioned in Memorandum Circular No. 2024-11 on Guidelines on the Issuance of CRDs)"
  - item: "Duly notarized Affidavit of Nearest Kin provided that they show link/relationship as the nearest surviving kin of the document owner, if applicable – one (1) original copy"
    whereToSecure: "Notary Public/ Nearest Kin"
  - item: "Duly notarized Affidavit of Guardianship, if applicable – one (1) original copy"
    whereToSecure: "Notary Public/ Legal guardian/ custodian"
  - item: "Supporting document for Affidavit of Guardianship: Barangay Certification that the minor is in the custody of the requesting party/guardian, if applicable"
    whereToSecure: "Office of the Punong Barangay"
  - item: "Supporting document for Affidavit of Guardianship: School ID/Report Card of the minor that indicates the name of the requester as Guardian – one (1) original and one (1) photocopy, if applicable"
    whereToSecure: "Educational institution recognized by the Department of Education"
  - item: "Letter of Request from court or proper public official whenever necessary in administrative, judicial or other official proceedings, if applicable – one (1) original copy"
    whereToSecure: "Court, proper public official or duly licensed and accredited agencies"
  - item: "Printed copy of Authorization Letters/SPA with picture of the Document Owner or Authorizing Person holding both the original Authorization Letter and Valid ID (for Authorization Letter sent via email or messenger), if applicable – one (1) copy"
    whereToSecure: "Document Owner/Authorizing Person"

    processingTime: "5 minutes (varying on client's filling-out of request form)"
    personResponsible: "Administrative Aide VI or Administrative Aide I"
  - clientStep: "Fill-out request form"
    agencyAction: "If the requesting party is the document owner, verifies the Valid ID presented"
    fee: "None"
    processingTime: "5 minutes (varying on client's filling-out of request form)"
    personResponsible: "Administrative Aide VI or Administrative Aide I"
  - clientStep: "Log personal information and purpose on the Client Log-book for Contact Trace"

    fee: "None"
    processingTime: "5 minutes (varying on client's filling-out of request form)"
    personResponsible: "Administrative Aide VI or Administrative Aide I"

    fee: "None"
    processingTime: "5 minutes (varying on client's filling-out of request form)"
    personResponsible: "Administrative Aide VI or Administrative Aide I"
  - clientStep: "Get queuing number and Citizens Charter (ARTA) Monitoring Form"
    agencyAction: "Provides queuing number and Citizens Charter ARTA Monitoring Form and informs client to submit queuing number and request form at Window 1"
    fee: "None"
    processingTime: "5 minutes (varying on client's filling-out of request form)"
    personResponsible: "Administrative Aide VI or Administrative Aide I"
  - clientStep: "Submit queuing number, Citizens Charter ARTA Monitoring Form and all necessary documents and requirements"
    agencyAction: "Receives queuing number, Citizens Charter (ARTA) Monitoring Form and filled-out Request Form"
    fee: "None"
    processingTime: "20 minutes (varying on the availability of documents to be searched)"
    personResponsible: "Administrative Aide VI or Administrative Assistant III"
  - clientStep: "Wait for notification"

    personResponsible: "Administrative Aide IV"
  - clientStep: "Wait for notification"
    agencyAction: "Forward the request form to the Digital Archiving Section and search through the Manage Point System and/or the Records and Archiving Section for a manual search of the document if it is not available in the database"
    fee: "None"
    processingTime: "20 minutes (varying on the availability of documents to be searched)"
    personResponsible: "Administrative Aide IV or Administrative Aide I"
  - clientStep: "Wait for notification"
    agencyAction: "If the document is available (positive) or not available (negative), issue the order of payment. Note: The office will issue certification regarding the availability and/or unavailability of the requested civil registry document"
    fee: "None"
    processingTime: "20 minutes (varying on the availability of documents to be searched)"
    personResponsible: "Administrative Aide IV or Administrative Aide I"
  - phase: "Payment of Applicable Fees at the City Treasurer's Office"
    clientStep: "Personal Information System (PIS) Verification. Without PIS, proceed to Management Information System Section. With PIS, proceed to City Treasurer's Office for payment of applicable fees"
    agencyAction: "Certified Photocopy (CPC)/Certified Transcription/Certification (both local purposes and abroad)"
    fee: "PHP 100.00 per copy"
    processingTime: "Not specified in the Citizen's Charter"
    personResponsible: "City Treasurer's Office"
  - clientStep: "Personal Information System (PIS) Verification. Without PIS, proceed to Management Information System Section. With PIS, proceed to City Treasurer's Office for payment of applicable fees"

    fee: "PHP 500.00 per document"
    processingTime: "Not specified in the Citizen's Charter"
    personResponsible: "City Treasurer's Office"
  - clientStep: "Personal Information System (PIS) Verification. Without PIS, proceed to Management Information System Section. With PIS, proceed to City Treasurer's Office for payment of applicable fees"
    agencyAction: "Documentary Stamp Tax"
    fee: "PHP 30.00 per document"
    processingTime: "Not specified in the Citizen's Charter"
    personResponsible: "City Treasurer's Office"

    agencyAction: "ICT Fee"

    processingTime: "Not specified in the Citizen's Charter"
    personResponsible: "City Treasurer's Office"

    agencyAction: "Receives Official Receipt, records its Unique Transaction Number (UTN) and Official Receipt Number"
    fee: "None"
    processingTime: "5 minutes (varying on the number of documents requested)"
    personResponsible: "Administrative Aide IV, Administrative Aide VI, Administrative Assistant III, Administrative Officer I, Registration Officer II, Registration Officer III, Registration Officer IV or City Civil Registrar"
  - clientStep: "Wait for notification"
    agencyAction: "Prepare the Certified Photocopy (CPC)/Certified Transcription/Certification of the requested document"
    fee: "None"
    processingTime: "5 minutes (varying on the number of documents requested)"
    personResponsible: "Administrative Aide IV, Administrative Aide VI, Administrative Assistant III, Administrative Officer I, Registration Officer II, Registration Officer III, Registration Officer IV or City Civil Registrar"

    agencyAction: "Verifier and Civil Registrar or authorized signatory signs the document"
    fee: "None"
    processingTime: "5 minutes (varying on the number of documents requested)"

    fee: "None"
    processingTime: "5 minutes"
    personResponsible: "Administrative Aide VI or Administrative Aide I"
